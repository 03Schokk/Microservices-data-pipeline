--------- 1. Пересобрать и запустить всё с нуля ---------
docker-compose down -v
docker-compose up -d

--------- 2. Получить пользовательский токен ---------
$token = (curl.exe -X POST http://localhost:8000/token -H "Content-Type: application/x-www-form-urlencoded" -d "grant_type=password&username=admin&password=secret" | ConvertFrom-Json).access_token

--------- 3. Запустить генерацию данных ---------
curl.exe -X POST http://localhost:8000/api/lab1/generate -H "Authorization: Bearer $token"

--------- 4. Запросить отчёт (термин "база", период сентябрь–декабрь 2024) ---------
$term = [uri]::EscapeDataString("база")
$url = "http://localhost:8000/api/lab1/report?term=$term&start_date=2024-09-01&end_date=2024-12-24"
curl.exe -X POST $url -H "Authorization: Bearer $token"
